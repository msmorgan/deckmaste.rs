fn surface_for_nouns(lexeme: Nouns, number: Number) -> &'static str {
    match (lexeme, number) {
        (Nouns::Person, Number::Singular) => "person",
        (Nouns::Person, Number::Plural) => "persons",
    }
}

    match (lexeme, agreement) {
        (Verbs::Act, Agreement::Bare) => "act",
        (Verbs::Act, Agreement::ThirdPersonSingular) => "acts",

}

    Static { kind: LexicalProvenanceKind, stable_id: &'static str },
    Declaration(
        std::sync::Arc<
            (
                ::macro_ron::v2::DeclarationIdentity,
                ::macro_ron::v2::SurfaceFeature,

            ),
        >,
    ),
}

    identity: LexicalOwnerIdentity,
}
impl Lexical {
    pub(crate) const fn class(self) -> TerminalClass {
        match self {
            Lexical::Literal(_) => unreachable!(),
            Lexical::EndOfInput => TerminalClass::EndOfInput,
            Lexical::Words => TerminalClass::Words,
            Lexical::Noun(_) => TerminalClass::Noun,
            Lexical::Verb(_) => TerminalClass::VerbLexeme,
            Lexical::SignedNumber => TerminalClass::SignedNumber,
            Lexical::Declaration(matcher) => {
                TerminalClass::Declaration(DeclarationClass {
                    kind: matcher.kind,
                    position: matcher.position,
                })
            }
        }
    }
}
impl LexicalTerminal {
    pub(crate) const fn class(self) -> TerminalClass {
        self.matcher.class()
    }
}
impl TerminalClass {
    pub const fn label(self) -> &'static str {
        match self {
            TerminalClass::EndOfInput => "end of input",
            TerminalClass::Words => "words",
            TerminalClass::Noun => "noun",
            TerminalClass::VerbLexeme => "verb lexeme",
            TerminalClass::SignedNumber => "signed number",
            TerminalClass::Declaration(_) => "open declaration",
        }
    }
}
impl std::fmt::Display for TerminalClass {
    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        formatter.write_str(self.label())
    }
}
impl DeclarationClass {
    pub const fn kind(self) -> ::macro_ron::v2::DeclarationKind {
        self.kind
    }
    pub const fn position(self) -> ::macro_ron::v2::GrammarPosition {
        self.position
    }
}
pub(crate) fn declaration_lexeme_owner_id(
    id: &::macro_ron::v2::DeclarationIdentity,
    feature: ::macro_ron::v2::SurfaceFeature,
) -> String {
    let kind = match id.kind() {
        ::macro_ron::v2::DeclarationKind::KeywordAction => "keyword_action",
        ::macro_ron::v2::DeclarationKind::KeywordAbility => "keyword_ability",
        ::macro_ron::v2::DeclarationKind::Subtype(category) => {
            match category {
                ::macro_ron::v2::SubtypeCategory::Artifact => "artifact_subtype",
                ::macro_ron::v2::SubtypeCategory::Battle => "battle_subtype",
                ::macro_ron::v2::SubtypeCategory::Creature => "creature_subtype",
                ::macro_ron::v2::SubtypeCategory::Enchantment => "enchantment_subtype",
                ::macro_ron::v2::SubtypeCategory::Land => "land_subtype",
                ::macro_ron::v2::SubtypeCategory::Planeswalker => "planeswalker_subtype",
                ::macro_ron::v2::SubtypeCategory::Spell => "spell_subtype",
            }
        }
        ::macro_ron::v2::DeclarationKind::Type => "type",
        ::macro_ron::v2::DeclarationKind::CounterKind => "counter_kind",
        ::macro_ron::v2::DeclarationKind::Designation => "designation",
    };
    let feature = match feature {
        ::macro_ron::v2::SurfaceFeature::Bare => "bare",
        ::macro_ron::v2::SurfaceFeature::ThirdPersonSingular => "third_person_singular",
        ::macro_ron::v2::SurfaceFeature::Singular => "singular",
        ::macro_ron::v2::SurfaceFeature::Plural => "plural",
        ::macro_ron::v2::SurfaceFeature::Fixed => "fixed",
    };
    LexicalOwner::construct_label(|| {
        format!("lexeme:{kind}/{}/{feature}", id.name())
    })
}
impl LexicalOwner {
    #[cfg(test)]
    fn update_label_constructions(set: Option<usize>, increment: bool) -> usize {
        std::thread_local! {
            static COUNT : std::cell::Cell < usize > = const { std::cell::Cell::new(0) };
        }
        COUNT
            .with(|count| {
                if let Some(value) = set {
                    count.set(value);
                }
                if increment {
                    count.set(count.get() + 1);
                }
                count.get()
            })
    }
    #[cfg(test)]
    pub(crate) fn reset_label_constructions() {
        Self::update_label_constructions(Some(0), false);
    }
    #[cfg(test)]
    pub(crate) fn label_constructions() -> usize {
        Self::update_label_constructions(None, false)
    }
    fn construct_label(build: impl FnOnce() -> String) -> String {
        #[cfg(test)] Self::update_label_constructions(None, true);
        build()
    }
    pub(crate) fn static_owner(
        kind: LexicalProvenanceKind,
        stable_id: &'static str,
    ) -> Self {
        Self {
            identity: LexicalOwnerIdentity::Static {
                kind,
                stable_id,
            },
        }
    }
    pub(crate) fn declaration_owner(
        id: ::macro_ron::v2::DeclarationIdentity,
        feature: ::macro_ron::v2::SurfaceFeature,
    ) -> Self {
        Self {
            identity: LexicalOwnerIdentity::Declaration(
                std::sync::Arc::new((id, feature, std::sync::OnceLock::new())),
            ),
        }
    }
    pub const fn kind(&self) -> LexicalProvenanceKind {
        match self.identity {
            LexicalOwnerIdentity::Static { kind, .. } => kind,
            LexicalOwnerIdentity::Declaration(_) => LexicalProvenanceKind::Lexeme,
        }
    }
    pub fn stable_id(&self) -> &str {
        match &self.identity {
            LexicalOwnerIdentity::Static { stable_id, .. } => stable_id,
            LexicalOwnerIdentity::Declaration(declaration) => {
                declaration
                    .2
                    .get_or_init(|| declaration_lexeme_owner_id(
                        &declaration.0,
                        declaration.1,
                    ))
            }
        }
    }
    pub(crate) fn stable_id_owned(&self) -> String {
        match &self.identity {
            LexicalOwnerIdentity::Static { stable_id, .. } => {
                Self::construct_label(|| stable_id.to_string())
            }
            LexicalOwnerIdentity::Declaration(_) => self.stable_id().to_owned(),
        }
    }
}
impl std::fmt::Debug for LexicalOwner {
    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        formatter
            .debug_struct("LexicalOwner")
            .field("kind", &self.kind())
            .field("stable_id", &self.stable_id())
            .finish()
    }
}
impl Clone for LexicalOwner {
    fn clone(&self) -> Self {
        Self {
            identity: self.identity.clone(),
        }
    }
}
impl PartialEq for LexicalOwner {
    fn eq(&self, other: &Self) -> bool {
        match (&self.identity, &other.identity) {
            (
                LexicalOwnerIdentity::Static { kind: left_kind, stable_id: left_id },
                LexicalOwnerIdentity::Static { kind: right_kind, stable_id: right_id },
            ) => left_kind == right_kind && left_id == right_id,
            (
                LexicalOwnerIdentity::Declaration(left),
                LexicalOwnerIdentity::Declaration(right),
            ) => left.0 == right.0 && left.1 == right.1,
            _ => false,
        }
    }
}
impl Eq for LexicalOwner {}
impl Ord for LexicalOwner {
    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.kind()
            .cmp(&other.kind())
            .then_with(|| {
                match (&self.identity, &other.identity) {
                    (
                        LexicalOwnerIdentity::Static { stable_id: left, .. },
                        LexicalOwnerIdentity::Static { stable_id: right, .. },
                    ) => left.cmp(right),
                    (
                        LexicalOwnerIdentity::Declaration(left),
                        LexicalOwnerIdentity::Declaration(right),
                    ) => left.0.cmp(&right.0).then_with(|| left.1.cmp(&right.1)),
                    (LexicalOwnerIdentity::Static { .. }, _) => std::cmp::Ordering::Less,
                    (LexicalOwnerIdentity::Declaration(_), _) => {
                        std::cmp::Ordering::Greater
                    }
                }
            })
    }
}
impl PartialOrd for LexicalOwner {
    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
        Some(self.cmp(other))
    }
}
impl LexicalOwnerTemplate {
    pub(crate) fn instantiate(self, value: &Leaf) -> Option<LexicalOwner> {
        match (self, value) {
            (LexicalOwnerTemplate::None, Leaf::EndOfInput) => None,
            (LexicalOwnerTemplate::Static { kind, stable_id }, _) => {
                Some(LexicalOwner::static_owner(kind, stable_id))
            }
            (
                LexicalOwnerTemplate::Vocab { declaration: "Words" },
                Leaf::Words(Words::First),
            ) => {
                Some(
                    LexicalOwner::static_owner(
                        LexicalProvenanceKind::Vocab,
                        "vocab:Words/First",
                    ),
                )
            }
            (
                LexicalOwnerTemplate::Vocab { declaration: "Words" },
                Leaf::Words(Words::Second),
            ) => {
                Some(
                    LexicalOwner::static_owner(
                        LexicalProvenanceKind::Vocab,
                        "vocab:Words/Second",
                    ),
                )
            }
            (LexicalOwnerTemplate::Lexeme { stable_id }, _) => {
                Some(
                    LexicalOwner::static_owner(LexicalProvenanceKind::Lexeme, stable_id),
                )
            }
            (
                LexicalOwnerTemplate::Declaration { kind, name },
                Leaf::Declaration(declaration),
            ) if declaration.id.kind() == kind && declaration.id.name() == name => {
                Some(
                    LexicalOwner::declaration_owner(
                        declaration.id.clone(),
                        declaration.feature,
                    ),
                )
            }
            _ => unreachable!("validated lexical owner template/value mismatch"),
        }
    }
}
pub(crate) fn scan_lexical(
    input: &ScanInput<'_>,
    terminal: LexicalTerminal,
) -> Vec<LexicalMatch<Leaf, LexicalOwner>> {
    let mut matches: Vec<LexicalMatch<Leaf, LexicalOwner>> = match terminal.matcher {
        Lexical::EndOfInput => {
            (input.position.byte_offset == input.text.len())
                .then_some(LexicalMatch {
                    end: input.position.byte_offset,
                    value: Leaf::EndOfInput,
                    owner: None,
                })
                .into_iter()
                .collect()
        }
        Lexical::Literal(literal @ (".")) => {
            input
                .punctuation_end(literal)
                .map(|end| LexicalMatch {
                    end,
                    value: Leaf::Literal(literal),
                    owner: None,
                })
                .into_iter()
                .collect()
        }
        Lexical::Literal(literal) => {
            input
                .word_end(literal)
                .map(|end| LexicalMatch {
                    end,
                    value: Leaf::Literal(literal),
                    owner: None,
                })
                .into_iter()
                .collect()
        }
        Lexical::Words => {
            [("first", Words::First), ("second", Words::Second)]
                .into_iter()
                .filter_map(|(running_text, value)| {
                    input
                        .word_end(running_text)
                        .map(|end| LexicalMatch {
                            end,
                            value: Leaf::Words(value),
                            owner: None,
                        })
                })
                .collect()
        }
        Lexical::SignedNumber => {
            let offset = input.position.byte_offset;
            let prefix = usize::from(input.position.case == CasePosition::Continuation);
            let Some(remainder) = input.text.get(offset..) else {
                return Vec::new();
            };
            let Some(number) = (prefix == 0)
                .then_some(remainder)
                .or_else(|| remainder.strip_prefix(' ')) else {
                return Vec::new();
            };
            let (sign, digits) = number
                .strip_prefix('-')
                .map_or((Sign::Positive, number), |digits| (Sign::Negative, digits));
            let digit_length = digits.bytes().take_while(u8::is_ascii_digit).count();
            let digits = &digits[..digit_length];
            let Some(magnitude) = (!digits.is_empty())
                .then(|| digits.parse::<u32>().ok())
                .flatten() else {
                return Vec::new();
            };
            let candidate_length = usize::from(sign == Sign::Negative) + digit_length;
            let candidate = &number[..candidate_length];
            (magnitude.to_string() == digits)
                .then(|| input.word_end(candidate))
                .flatten()
                .map(|end| LexicalMatch {
                    end,
                    value: Leaf::SignedNumber(SignedNumber { sign, magnitude }),
                    owner: None,
                })
                .into_iter()
                .collect()
        }
        Lexical::Noun(_) | Lexical::Verb(_) => scan_bound_terminal(input, terminal),
        Lexical::Declaration(matcher) => {
            input
                .declaration_readings(matcher)
                .into_iter()
                .map(|(end, id, feature)| LexicalMatch {
                    end,
                    value: Leaf::Declaration(DeclarationLeaf { id, feature }),
                    owner: None,
                })
                .collect()
        }
    };
    for lexical_match in &mut matches {
        lexical_match.owner = terminal.owner.instantiate(&lexical_match.value);
    }
    matches
}
impl Action {
    fn write_action_render(&self, writer: &mut Writer, context: &ParseContext<'_>) {
        match self {
            Self::Action(ActionElement { node }) => {
                writer
                    .claim(
                        || LexicalOwner::static_owner(
                            LexicalProvenanceKind::Lexeme,
                            "lexeme:Verbs/Act",
                        ),
                        |writer| {
                            writer.word(surface_for_verbs(Verbs::Act, Agreement::Bare));
                        },
                    );
                render_node(writer, node);
            }
        }
        writer
            .claim(
                || LexicalOwner::static_owner(
                    LexicalProvenanceKind::FormLiteral,
                    "root:Action/punctuation",
                ),
                |writer| writer.punctuation('.'),
            );
    }
}
impl Render for Action {
    fn render(&self, context: &ParseContext<'_>) -> String {
        let mut writer = Writer::new();
        self.write_action_render(&mut writer, context);
        writer.finish()
    }
}
pub(crate) fn render_action_with_claims(
    value: &Action,
    context: &ParseContext<'_>,
) -> (String, Vec<RawRenderedClaim>) {
    let mut claims = Vec::new();
    let mut writer = Writer::collecting(&mut claims);
    value.write_action_render(&mut writer, context);
    let rendered = writer.finish();
    (rendered, claims)
}
fn render_node(writer: &mut Writer, node: &Node) {
    match node {
        Node::Leaf(WordLeaf { word }) => {
            writer
                .claim(
                    || match *word {
                        Words::First => {
                            LexicalOwner::static_owner(
                                LexicalProvenanceKind::Vocab,
                                "vocab:Words/First",
                            )
                        }
                        Words::Second => {
                            LexicalOwner::static_owner(
                                LexicalProvenanceKind::Vocab,
                                "vocab:Words/Second",
                            )
                        }
                    },
                    |writer| {
                        render_words(writer, *word);
                    },
                )
        }
        Node::Chain(chain) => {
            render_node(writer, chain.next());
            writer
                .claim(
                    || match *&chain.word {
                        Words::First => {
                            LexicalOwner::static_owner(
                                LexicalProvenanceKind::Vocab,
                                "vocab:Words/First",
                            )
                        }
                        Words::Second => {
                            LexicalOwner::static_owner(
                                LexicalProvenanceKind::Vocab,
                                "vocab:Words/Second",
                            )
                        }
                    },
                    |writer| {
                        render_words(writer, *&chain.word);
                    },
                );
        }
    }
}
fn render_words(writer: &mut Writer, words: Words) {
    match words {
        Words::First => writer.word("first"),
        Words::Second => writer.word("second"),
    }
}
fn render_signed_number(writer: &mut Writer, number: &SignedNumber) {
    let magnitude = number.magnitude.to_string();
    match number.sign {
        Sign::Positive => writer.word(&magnitude),
        Sign::Negative => writer.word(&format!("-{magnitude}")),
    }
}
pub trait Visitor {
    fn visit_node(&mut self, node: &Node) {
        walk_node(self, node);
    }
    fn visit_action(&mut self, action: &Action) {
        walk_action(self, action);
    }
    fn visit_word_leaf(&mut self, word_leaf: &WordLeaf) {
        walk_word_leaf(self, word_leaf);
    }
    fn visit_chain(&mut self, chain: &Chain) {
        walk_chain(self, chain);
    }
    fn visit_action_element(&mut self, action_element: &ActionElement) {
        walk_action_element(self, action_element);
    }
    fn visit_words(&mut self, _words: Words) {}
    fn visit_nouns(&mut self, _nouns: Nouns) {}
    fn visit_verbs(&mut self, _verbs: Verbs) {}
    fn visit_sign(&mut self, _sign: Sign) {}
    fn visit_signed_number(&mut self, _number: &SignedNumber) {}
}
pub fn walk_node<V: Visitor + ?Sized>(visitor: &mut V, node: &Node) {
    match node {
        Node::Leaf(word_leaf) => visitor.visit_word_leaf(word_leaf),
        Node::Chain(chain) => visitor.visit_chain(chain),
    }
}
pub fn walk_action<V: Visitor + ?Sized>(visitor: &mut V, action: &Action) {
    match action {
        Action::Action(action_element) => visitor.visit_action_element(action_element),
    }
}
pub fn walk_word_leaf<V: Visitor + ?Sized>(visitor: &mut V, word_leaf: &WordLeaf) {
    let WordLeaf { word } = word_leaf;
    walk_words(visitor, *word);
}
pub fn walk_chain<V: Visitor + ?Sized>(visitor: &mut V, chain: &Chain) {
    visitor.visit_node(chain.next());
    walk_words(visitor, *&chain.word);
}
pub fn walk_action_element<V: Visitor + ?Sized>(
    visitor: &mut V,
    action_element: &ActionElement,
) {
    let ActionElement { node } = action_element;
    walk_verbs(visitor, Verbs::Act);
    visitor.visit_node(node);
}
pub fn walk_words<V: Visitor + ?Sized>(visitor: &mut V, words: Words) {
    match words {
        Words::First => visitor.visit_words(Words::First),
        Words::Second => visitor.visit_words(Words::Second),
    }
}
pub fn walk_nouns<V: Visitor + ?Sized>(visitor: &mut V, nouns: Nouns) {
    match nouns {
        Nouns::Person => visitor.visit_nouns(Nouns::Person),
    }
}
pub fn walk_verbs<V: Visitor + ?Sized>(visitor: &mut V, verbs: Verbs) {
    match verbs {
        Verbs::Act => visitor.visit_verbs(Verbs::Act),
    }
}
pub fn walk_sign<V: Visitor + ?Sized>(visitor: &mut V, sign: Sign) {
    visitor.visit_sign(sign);
}
pub fn walk_signed_number<V: Visitor + ?Sized>(visitor: &mut V, number: &SignedNumber) {
    visitor.visit_sign(number.sign);
    visitor.visit_signed_number(number);
}

#[repr(usize)]
#[derive(Debug, Clone, Copy, PartialEq, Eq, Ord, PartialOrd)]
pub(crate) enum RuleId {
    NodeLeaf,
    NodeChain,
    ActionAction,
}
impl RuleId {
    #[cfg(test)]
    pub(crate) const COUNT: usize = 3;
    pub(crate) const fn construction(self) -> Construction {
        match self {
            Self::NodeLeaf => Construction::NodeLeaf,
            Self::NodeChain => Construction::NodeChain,
            Self::ActionAction => Construction::ActionAction,
        }
    }
    pub(crate) const fn index(self) -> usize {
        self as usize
    }
}
pub(crate) const RULES: &[Rule<Category, LexicalTerminal, RuleId>] = &[
    Rule {
        id: RuleId::NodeLeaf,
        lhs: Category::Node,
        rhs: &[
            L(LexicalTerminal {
                matcher: Lexical::Words,
                owner: LexicalOwnerTemplate::Vocab {
                    declaration: "Words",
                },
            }),
        ],
    },
    Rule {
        id: RuleId::NodeChain,
        lhs: Category::Node,
        rhs: &[
            N(Category::Node),
            L(LexicalTerminal {
                matcher: Lexical::Words,
                owner: LexicalOwnerTemplate::Vocab {
                    declaration: "Words",
                },
            }),
        ],
    },
    Rule {
        id: RuleId::ActionAction,
        lhs: Category::Action,
        rhs: &[
            L(LexicalTerminal {
                matcher: Lexical::Verb(Verbs::Act),
                owner: LexicalOwnerTemplate::Lexeme {
                    stable_id: "lexeme:Verbs/Act",
                },
            }),
            N(Category::Node),
            L(LexicalTerminal {
                matcher: Lexical::Literal("."),
                owner: LexicalOwnerTemplate::Static {
                    kind: LexicalProvenanceKind::FormLiteral,
                    stable_id: "root:Action/punctuation",
                },
            }),
            L(LexicalTerminal {
                matcher: Lexical::EndOfInput,
                owner: LexicalOwnerTemplate::None,
            }),
        ],
    },
];
#[expect(
    clippy::too_many_lines,
    reason = "the exhaustive generated-shape construction dispatch is intentionally flat"
)]
pub(super) fn build(
    rule: RuleId,
    children: &[BuildValue],
    context: &ParseContext<'_>,
) -> Option<BuildValue> {
    match rule {
        RuleId::NodeLeaf => {
            match children {
                [BuildValue::Leaf(Leaf::Words(words))] => {
                    Some(BuildValue::Node(Node::Leaf(WordLeaf { word: *words })))
                }
                _ => None,
            }
        }
        RuleId::NodeChain => {
            match children {
                [BuildValue::Node(next), BuildValue::Leaf(Leaf::Words(words))] => {
                    Chain::new(Box::new(next.clone()), *words)
                        .map(Node::Chain)
                        .map(BuildValue::Node)
                }
                _ => None,
            }
        }
        RuleId::ActionAction => {
            match children {
                [BuildValue::Leaf(
                    Leaf::Verb { lexeme: Verbs::Act, agreement: Agreement::Bare },
                ), BuildValue::Node(node), BuildValue::Leaf(Leaf::Literal(".")),
                BuildValue::Leaf(Leaf::EndOfInput)] => {
                    Some(
                        BuildValue::Action(
                            Action::Action(ActionElement {
                                node: node.clone(),
                            }),
                            Agreement::Bare,
                        ),
                    )
                }
                _ => None,
            }
        }
    }
}
