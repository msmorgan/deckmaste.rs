/// Values in declared slot order, with licensed intervening Adjuncts. No program or
/// production identity is stored here: the selected Word retains its local
/// declared frame choice and lexical owner.
#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
pub enum FrameValue {
    Argument(Box<Reading>),
    /// The Object stays in its declared slot; this value retains its
    /// postposed Linearization and any preceding Depictive Adjunct.
    PostposedObject {
        argument: Box<Reading>,
        modifier: Option<Box<Reading>>,
    },
    Adjunct(Box<Reading>),
    Marker(Word),
    Marked {
        marker: Word,
        argument: Box<Reading>,
    },
    Optional(Option<Box<FrameValue>>),
}

fn frame_values_in_surface_order(values: &[FrameValue]) -> impl Iterator<Item = &FrameValue> {
    values
        .iter()
        .filter(|value| !value.is_postposed())
        .chain(values.iter().filter(|value| value.is_postposed()))
}

#[derive(Clone, Copy)]
enum FrameHead<'a> {
    Lexical(&'a Word),
    Constituent(&'a Reading),
}

#[derive(Clone)]
enum FrameGuard {
    Any,
    Features(Vec<(usize, FeatureValue)>),
    Head(usize),
    Marker(usize, Option<(usize, TableOperation, FeatureValue)>),
}

impl FrameGuard {
    fn admits(&self, child: Option<&Summary>) -> bool {
        match self {
            Self::Any => true,
            Self::Features(requirements) => child.is_some_and(|summary| {
                requirements.iter().all(|(feature, expected)| {
                    summary.values.get(*feature).and_then(Option::as_ref) == Some(expected)
                })
            }),
            Self::Head(frame) => child.is_some_and(|summary| {
                summary.values.get(7).and_then(Option::as_ref) == Some(&FeatureValue::Frame(*frame))
            }),
            Self::Marker(class, licence) => child.is_some_and(|summary| {
                summary.marker_class == Some(*class)
                    && licence
                        .as_ref()
                        .is_none_or(|(feature, operation, expected)| {
                            summary
                                .values
                                .get(*feature)
                                .and_then(Option::as_ref)
                                .and_then(|value| operation.apply(::std::slice::from_ref(value)))
                                .as_ref()
                                == Some(expected)
                        })
            }),
        }
    }
}

#[derive(Clone)]
enum FrameLayout {
    Argument(Category, Vec<(usize, FeatureValue)>),
    PostposedObject(Category, Vec<(usize, FeatureValue)>, Option<Category>),
    Adjunct(Category),
    Marker(usize, ::deckmaste_lexical::Category),
    Marked(
        usize,
        ::deckmaste_lexical::Category,
        Category,
        Vec<(usize, FeatureValue)>,
    ),
    Optional(Option<Box<FrameLayout>>),
}

// Open frame dependencies contain candidates for each segment, never AST products.
// The first segment pins the overt head's frame choice; other segments may use
// corresponding slots in a different frame declared by that same lexical owner.
#[derive(Clone)]
enum SegmentRule {
    None,
    Build {
        kind: String,
        candidates: Vec<(usize, Vec<(usize, FrameGuard)>)>,
    },
    Share(usize, usize),
    Discharge(usize, usize),
}

impl SegmentRule {
    fn advance(&self, dot: usize, state: &mut State, child: Option<&Summary>) -> Option<()> {
        let open = child.is_some_and(|summary| !summary.segments.is_empty());
        match self {
            Self::None => {
                if open {
                    return None;
                }
            }
            Self::Build { candidates, .. } => {
                if open {
                    return None;
                }
                if dot == 0 {
                    state.segment_candidates = (0..candidates.len()).collect();
                }
                state.segment_candidates.retain(|&index| {
                    candidates[index]
                        .1
                        .iter()
                        .all(|(slot, guard)| *slot != dot || guard.admits(child))
                });
                if state.segment_candidates.is_empty() {
                    return None;
                }
            }
            Self::Share(left, right) => {
                if dot == *left || dot == *right {
                    let child = child?;
                    if child.segments.is_empty() {
                        return None;
                    }
                    state.segments.extend(child.segments.iter().cloned());
                } else if open {
                    return None;
                }
            }
            Self::Discharge(head, tail) => {
                if dot == *head {
                    let child = child?;
                    let Some(FeatureValue::Frame(selected)) = child.values.get(7)?.as_ref() else {
                        return None;
                    };
                    state.segment_license = Some((*selected, child.licensed_segments.clone()));
                } else if dot == *tail {
                    let child = child?;
                    let (selected, licensed) = state.segment_license.as_ref()?;
                    if !child.segments.first()?.contains(selected)
                        || child.segments.iter().any(|candidates| {
                            !candidates
                                .iter()
                                .any(|candidate| licensed.contains(candidate))
                        })
                    {
                        return None;
                    }
                    state.segment_license = None;
                } else if open {
                    return None;
                }
            }
        }
        Some(())
    }

    fn complete(&self, state: &State) -> Option<Vec<Vec<usize>>> {
        match self {
            Self::Build { candidates, .. } => {
                let mut frames: Vec<_> = state
                    .segment_candidates
                    .iter()
                    .map(|&index| candidates[index].0)
                    .collect();
                frames.sort_unstable();
                frames.dedup();
                if frames.is_empty() { None } else { Some(vec![frames]) }
            }
            Self::Share(_, _) => {
                if state.segments.len() < 2 {
                    return None;
                }
                // Repetition after the first imposes the same licensing obligation.
                // Keep the first distinct for the selected head's frame identity.
                let mut result = vec![state.segments[0].clone()];
                for segment in &state.segments[1..] {
                    if !result[1..].contains(segment) {
                        result.push(segment.clone());
                    }
                }
                result[1..].sort();
                Some(result)
            }
            Self::Discharge(_, _) if state.segment_license.is_some() => None,
            _ => Some(vec![]),
        }
    }
}

fn corresponding_frame(
    left: &::deckmaste_lexical::Frame,
    right: &::deckmaste_lexical::Frame,
) -> bool {
    fn corresponding_item(
        left: &::deckmaste_lexical::FrameItem,
        right: &::deckmaste_lexical::FrameItem,
    ) -> bool {
        use ::deckmaste_lexical::FrameItem;
        match (left, right) {
            (FrameItem::Argument(left), FrameItem::Argument(right)) => left == right,
            (
                FrameItem::Marker {
                    vocabulary: left, ..
                },
                FrameItem::Marker {
                    vocabulary: right, ..
                },
            ) => left == right,
            (
                FrameItem::Marked {
                    vocabulary: left,
                    slot: left_slot,
                    ..
                },
                FrameItem::Marked {
                    vocabulary: right,
                    slot: right_slot,
                    ..
                },
            ) => left == right && left_slot == right_slot,
            (FrameItem::Optional(left), FrameItem::Optional(right)) => {
                corresponding_item(left, right)
            }
            _ => false,
        }
    }
    left.kind == right.kind
        && left.items.len() == right.items.len()
        && left
            .items
            .iter()
            .zip(&right.items)
            .all(|(left, right)| corresponding_item(left, right))
}

struct FrameExpansion {
    template: usize,
    layout: Vec<FrameLayout>,
    order: Vec<usize>,
}

/// A declared frame excluded from compilation because its slot vocabulary has
/// not been reconciled with the grammar. It is never treated as an arbitrary NP.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct UnsupportedFrame {
    pub frame: ::deckmaste_lexical::Frame,
    pub reason: String,
}

/// One immutable lexicon/grammar pair, reused for recognition, checking,
/// realization and traversal. Compiled program identities are local to it.
pub struct GrammarEnvironment<'a> {
    lexicon: &'a ::deckmaste_lexical::Lexicon,
    grammar: Grammar,
}

impl<'a> GrammarEnvironment<'a> {
    pub fn new(lexicon: &'a ::deckmaste_lexical::Lexicon) -> Self {
        let grammar = Grammar::with_lexicon(lexicon);
        Self { lexicon, grammar }
    }
    pub fn grammar(&self) -> &Grammar {
        &self.grammar
    }
    pub fn unsupported_frames(&self) -> &[UnsupportedFrame] {
        &self.grammar.unsupported
    }
    pub fn parse(
        &self,
        text: &str,
        category: Category,
    ) -> Result<::deckmaste_english_v3::GrammarForest<Grammar>, ::deckmaste_english_v3::ParseError>
    {
        ::deckmaste_english_v3::parse(
            &self.grammar,
            self.lexicon,
            &self.lexicon.analyze(text),
            &category,
        )
    }
    pub fn admit(&self, reading: &Reading) -> Result<Summary, Error> {
        let summary = reading.admit_with(&self.grammar, self.lexicon)?;
        if !summary.segments.is_empty() {
            return Err(Error::Invalid("frame segments", "undischarged sharing"));
        }
        let mut output = Realization::new();
        reading.write(
            &self.grammar,
            self.lexicon,
            &mut output,
            &mut AdmissionCache::default(),
        )?;
        output.check_context(self.lexicon)?;
        Ok(summary)
    }
    pub fn realize(&self, reading: &Reading) -> Result<String, Error> {
        let summary = reading.admit_with(&self.grammar, self.lexicon)?;
        if !summary.segments.is_empty() {
            return Err(Error::Invalid("frame segments", "undischarged sharing"));
        }
        let mut output = Realization::new();
        reading.write(
            &self.grammar,
            self.lexicon,
            &mut output,
            &mut AdmissionCache::default(),
        )?;
        output.check_context(self.lexicon)?;
        Ok(output.finish())
    }
    pub fn visit(
        &self,
        reading: &Reading,
        visitor: &mut impl FnMut(&Reading),
    ) -> Result<(), Error> {
        self.admit(reading)?;
        reading.visit(visitor)
    }
    pub fn visit_words(
        &self,
        reading: &Reading,
        visitor: &mut impl FnMut(&Word),
    ) -> Result<(), Error> {
        self.admit(reading)?;
        reading.visit_words(visitor)
    }
}

impl Grammar {
    pub fn with_lexicon(lexicon: &::deckmaste_lexical::Lexicon) -> Self {
        let mut grammar = Self::default();
        grammar.compile_frames(lexicon);
        grammar
    }

    /// Check a reading against this compiled grammar and its supplied lexicon.
    pub fn admit(
        &self,
        reading: &Reading,
        lexicon: &::deckmaste_lexical::Lexicon,
    ) -> Result<Summary, Error> {
        if REQUIRES_ENVIRONMENT && !self.compiled {
            return Err(Error::EnvironmentRequired);
        }
        let summary = reading.admit_with(self, lexicon)?;
        if !summary.segments.is_empty() {
            return Err(Error::Invalid("frame segments", "undischarged sharing"));
        }
        let mut output = Realization::new();
        reading.write(self, lexicon, &mut output, &mut AdmissionCache::default())?;
        output.check_context(lexicon)?;
        Ok(summary)
    }
    /// Realize using the same compiled programs that recognized the reading.
    pub fn realize(
        &self,
        reading: &Reading,
        lexicon: &::deckmaste_lexical::Lexicon,
    ) -> Result<String, Error> {
        let output = self.realization(reading, lexicon)?;
        output.check_context(lexicon)?;
        Ok(output.finish())
    }
    fn realization(
        &self,
        reading: &Reading,
        lexicon: &::deckmaste_lexical::Lexicon,
    ) -> Result<Realization, Error> {
        self.realization_with_cache(reading, lexicon, &mut AdmissionCache::default())
    }
    fn realization_with_cache(
        &self,
        reading: &Reading,
        lexicon: &::deckmaste_lexical::Lexicon,
        cache: &mut AdmissionCache,
    ) -> Result<Realization, Error> {
        if REQUIRES_ENVIRONMENT && !self.compiled {
            return Err(Error::EnvironmentRequired);
        }
        let summary = reading.admit_with_cache(self, lexicon, cache)?;
        if !summary.segments.is_empty() {
            return Err(Error::Invalid("frame segments", "undischarged sharing"));
        }
        let mut output = Realization::new();
        reading.write(self, lexicon, &mut output, cache)?;
        Ok(output)
    }
    fn project(&self, features: ::deckmaste_english_v3::LexicalFeatures<'_>) -> Vec<Summary> {
        let properties = match &features {
            ::deckmaste_english_v3::LexicalFeatures::Word { properties, .. } => Some(*properties),
            _ => None,
        };
        let mut summaries = project(features);
        if let Some(properties) = properties {
            let marker = properties.features.get("FrameMarker").and_then(|class| {
                self.marker_classes
                    .iter()
                    .position(|(candidate, _)| candidate == class.as_str())
            });
            for summary in &mut summaries {
                summary.marker_class = marker;
                let frame = summary
                    .frame_choice
                    .and_then(|index| properties.frames.get(index));
                summary.values[7] = frame
                    .and_then(|frame| {
                        self.canonical_frames
                            .iter()
                            .position(|candidate| candidate == frame)
                    })
                    .map(FeatureValue::Frame);
                summary.licensed_segments = frame
                    .map(|selected| {
                        properties
                            .frames
                            .iter()
                            .filter(|candidate| {
                                if properties
                                    .features
                                    .get("SelectedTailLicence")
                                    .is_some_and(|licence| licence == "DeclaredFrames")
                                {
                                    candidate.kind == selected.kind
                                } else {
                                    corresponding_frame(selected, candidate)
                                }
                            })
                            .filter_map(|candidate| {
                                self.canonical_frames
                                    .iter()
                                    .position(|frame| frame == candidate)
                            })
                            .collect()
                    })
                    .unwrap_or_default();
            }
        }
        summaries
    }

    fn marker_class(
        &self,
        vocabulary: &str,
        member: &str,
    ) -> Result<(usize, ::deckmaste_lexical::Category), String> {
        let name = format!("{vocabulary}/{member}");
        let mut matches = self
            .marker_classes
            .iter()
            .enumerate()
            .filter(|(_, (candidate, _))| candidate == &name);
        let (index, (_, category)) = matches
            .next()
            .ok_or_else(|| format!("missing declared marker class {name}"))?;
        if matches.next().is_some() {
            return Err(format!("ambiguous declared marker class {name}"));
        }
        Ok((index, *category))
    }

    fn item_layouts(
        &mut self,
        item: &::deckmaste_lexical::FrameItem,
    ) -> Result<Vec<FrameLayout>, String> {
        use ::deckmaste_lexical::FrameItem;
        match item {
            FrameItem::Argument(slot) => Self::frame_category(&slot.category)
                .map(|category| {
                    vec![FrameLayout::Argument(
                        category,
                        frame_argument_requirements(&slot.category),
                    )]
                })
                .ok_or_else(|| {
                    format!(
                        "unsupported slot category {} ({:?})",
                        slot.category, slot.relation
                    )
                }),
            FrameItem::Marker { vocabulary, member } => {
                let (class, pos) = self.marker_class(vocabulary, member)?;
                Ok(vec![FrameLayout::Marker(class, pos)])
            }
            FrameItem::Marked {
                vocabulary,
                member,
                slot,
            } => {
                let category = Self::frame_category(&slot.category).ok_or_else(|| {
                    format!(
                        "unsupported slot category {} ({:?})",
                        slot.category, slot.relation
                    )
                })?;
                let (class, pos) = self.marker_class(vocabulary, member)?;
                Ok(vec![FrameLayout::Marked(
                    class,
                    pos,
                    category,
                    frame_argument_requirements(&slot.category),
                )])
            }
            FrameItem::Optional(item) => {
                let mut layouts = vec![FrameLayout::Optional(None)];
                layouts.extend(
                    self.item_layouts(item)?
                        .into_iter()
                        .map(|item| FrameLayout::Optional(Some(Box::new(item)))),
                );
                Ok(layouts)
            }
            FrameItem::Literal(text) => Err(format!("unreconciled legacy frame item {text}")),
        }
    }

    fn compile_frames(&mut self, lexicon: &::deckmaste_lexical::Lexicon) {
        self.compiled = true;
        self.static_count = self.productions.len();
        if SELECTED_FRAME_TEMPLATES.is_empty()
            && !self
                .rules
                .iter()
                .any(|rule| matches!(rule.segments, SegmentRule::Build { .. }))
        {
            return;
        }
        self.marker_classes = lexicon
            .lexemes()
            .values()
            .filter_map(|lexeme| {
                lexeme
                    .properties
                    .features
                    .get("FrameMarker")
                    .map(|class| (class.to_string(), lexeme.category))
            })
            .collect();
        self.marker_classes.sort();
        self.marker_classes.dedup();
        let mut frames = Vec::new();
        for lexeme in lexicon
            .lexemes()
            .values()
            .filter(|lexeme| lexeme.category == ::deckmaste_lexical::Category::Verb)
        {
            for frame in &lexeme.properties.frames {
                if !frames.contains(frame) {
                    frames.push(frame.clone());
                }
            }
        }
        for frame in &frames {
            if !self.canonical_frames.contains(frame) {
                self.canonical_frames.push(frame.clone());
            }
        }
        for frame in frames {
            if !SELECTED_FRAME_TEMPLATES
                .iter()
                .any(|(_, kind)| frame.kind == *kind) && !self.rules.iter().any(|rule| matches!(&rule.segments, SegmentRule::Build { kind, .. } if frame.kind == kind.as_str()))
            {
                continue;
            }
            let mut variants = vec![Vec::new()];
            let mut unsupported = None;
            for (position, item) in frame.items.iter().enumerate() {
                // A declared slot role admits an intervening Adjunct only
                // after an overt Object, independently of the selecting head.
                if let ::deckmaste_lexical::FrameItem::Argument(slot) = item {
                    if slot.relation == ::deckmaste_lexical::Relation::Complement
                        && position.checked_sub(1).is_some_and(|previous| {
                            matches!(&frame.items[previous],
                                ::deckmaste_lexical::FrameItem::Argument(previous)
                                if previous.relation == ::deckmaste_lexical::Relation::Object)
                        })
                    {
                        if let Some(category) = Self::frame_adjunct(&slot.category) {
                            let mut with_adjunct = variants.clone();
                            for prefix in &mut with_adjunct {
                                prefix.push(FrameLayout::Adjunct(category));
                            }
                            variants.extend(with_adjunct);
                        }
                    }
                }
                match self.item_layouts(item) {
                    Ok(options) => {
                        variants = variants
                            .into_iter()
                            .flat_map(|prefix| {
                                options.iter().map(move |option| {
                                    let mut result = prefix.clone();
                                    result.push(option.clone());
                                    result
                                })
                            })
                            .collect()
                    }
                    Err(reason) => {
                        unsupported = Some(reason);
                        break;
                    }
                }
            }
            if let Some(reason) = unsupported {
                self.unsupported.push(UnsupportedFrame { frame, reason });
                continue;
            }
            let program = self.programs.len();
            self.programs.push(frame);
            for index in 0..self.static_count {
                let SegmentRule::Build { kind, .. } = &self.rules[index].segments else {
                    continue;
                };
                if self.programs[program].kind != kind.as_str() {
                    continue;
                }
                for layout in &variants {
                    let mut shape = vec![];
                    let mut rule = self.rules[index].clone();
                    rule.guards.clear();
                    rule.literals.clear();
                    rule.checks.clear();
                    rule.lexical_checks.clear();
                    rule.release.clear();
                    for item in layout {
                        item.symbols(&mut shape, &mut rule, &self.programs[program].kind);
                    }
                    // A segment omits the selecting head and its preceding separator.
                    if !shape.is_empty() {
                        shape.remove(0);
                        rule.guards.remove(0);
                        rule.literals.remove(0);
                    }
                    let matches = shape.len() == self.productions[index].symbols.len()
                        && shape.iter().zip(&self.productions[index].symbols).all(
                            |(left, right)| {
                                use ::deckmaste_english_v3::Symbol;
                                match (left, right) {
                                    (Symbol::Literal(left), Symbol::Literal(right)) => {
                                        left == right
                                    }
                                    (Symbol::Lexical(left), Symbol::Lexical(right)) => {
                                        left == right
                                    }
                                    (Symbol::Nonterminal(left), Symbol::Nonterminal(right)) => {
                                        left == right
                                    }
                                    _ => false,
                                }
                            },
                        );
                    if matches {
                        let canonical = self
                            .canonical_frames
                            .iter()
                            .position(|frame| frame == &self.programs[program])
                            .unwrap();
                        let requirements = rule
                            .guards
                            .iter()
                            .enumerate()
                            .filter_map(|(slot, guard)| {
                                if matches!(guard, FrameGuard::Any) {
                                    None
                                } else {
                                    Some((slot, guard.clone()))
                                }
                            })
                            .collect();
                        let SegmentRule::Build { candidates, .. } = &mut self.rules[index].segments
                        else {
                            unreachable!()
                        };
                        candidates.push((canonical, requirements));
                    }
                }
            }
            for &(template, kind) in SELECTED_FRAME_TEMPLATES {
                if self.programs[program].kind != kind {
                    continue;
                }
                let layouts = self.frame_orders(&self.programs[program], &variants, lexicon);
                for (layout, order) in layouts {
                    let original = &self.rules[template];
                    let mut rule = original.clone();
                    rule.checks.truncate(1);
                    rule.lexical_checks.truncate(1);
                    rule.literals.truncate(1);
                    rule.release.truncate(1);
                    let canonical = self
                        .canonical_frames
                        .iter()
                        .position(|frame| frame == &self.programs[program])
                        .unwrap();
                    rule.guards = vec![FrameGuard::Head(canonical)];
                    let mut symbols = vec![self.productions[template].symbols[0].clone()];
                    for &index in &order {
                        layout[index].symbols(
                            &mut symbols,
                            &mut rule,
                            &self.programs[program].kind,
                        );
                    }
                    self.productions.push(::deckmaste_english_v3::Production {
                        category: self.productions[template].category,
                        symbols,
                    });
                    self.rules.push(rule);
                    self.expansion_index
                        .entry((template, program))
                        .or_default()
                        .push(self.expansions.len());
                    self.expansions.push(FrameExpansion {
                        template,
                        layout,
                        order,
                    });
                }
            }
        }
        self.frame_programs = self
            .canonical_frames
            .iter()
            .map(|frame| self.programs.iter().position(|program| program == frame))
            .collect();
    }

    fn frame_orders(
        &self,
        frame: &::deckmaste_lexical::Frame,
        variants: &[Vec<FrameLayout>],
        lexicon: &::deckmaste_lexical::Lexicon,
    ) -> Vec<(Vec<FrameLayout>, Vec<usize>)> {
        use ::deckmaste_lexical::{Category as LexicalCategory, FrameItem, Relation};
        fn object(item: &FrameItem) -> bool {
            match item {
                FrameItem::Argument(slot) => slot.relation == Relation::Object,
                FrameItem::Optional(item) => object(item),
                _ => false,
            }
        }
        let mut result: Vec<_> = variants
            .iter()
            .map(|layout| (layout.clone(), (0..layout.len()).collect()))
            .collect();
        let Some((object_category, marked_categories, modifier)) = frame_postposing(&frame.kind)
        else {
            return result;
        };
        // Only the unmarked Object is moved. A marker's nominal Complement
        // may itself have a legacy Object slot label, but is not another Object.
        for (slot_index, _) in frame
            .items
            .iter()
            .enumerate()
            .filter(|(_, item)| object(item))
        {
            for layout in variants {
                let index = layout
                    .iter()
                    .enumerate()
                    .filter(|(_, item)| !matches!(item, FrameLayout::Adjunct(_)))
                    .nth(slot_index)
                    .map(|(index, _)| index)
                    .expect("one layout per declared frame slot");
                let previous = layout[..index].iter().rev().find_map(FrameLayout::overt_slot);
                if matches!(
                    previous,
                    Some(FrameLayout::Marker(class, LexicalCategory::Preposition))
                        if self.marker_takes_complement(*class, lexicon)
                ) {
                    continue;
                }
                let mut following = layout[index + 1..].iter().filter_map(FrameLayout::overt_slot);
                let marked = match following.next() {
                    Some(FrameLayout::Marker(class, LexicalCategory::Preposition)) => {
                        self.marker_takes_complement(*class, lexicon)
                            && matches!(following.next(), Some(FrameLayout::Argument(_, _)))
                    }
                    Some(FrameLayout::Marked(class, LexicalCategory::Preposition, _, _)) => {
                        self.marker_takes_complement(*class, lexicon)
                    }
                    Some(FrameLayout::Argument(category, _)) => {
                        marked_categories.contains(category)
                    }
                    _ => false,
                };
                if !marked {
                    continue;
                }
                let order: Vec<_> = (0..layout.len())
                    .filter(|candidate| *candidate != index)
                    .chain([index])
                    .collect();
                for preceding in ::std::iter::once(None).chain(modifier.map(Some)) {
                    let Some(object) = layout[index]
                        .postposed_object(preceding, &frame_argument_requirements(object_category))
                    else {
                        continue;
                    };
                    let mut postposed = layout.clone();
                    postposed[index] = object;
                    result.push((postposed, order.clone()));
                }
            }
        }
        result
    }

    fn marker_takes_complement(
        &self,
        class: usize,
        lexicon: &::deckmaste_lexical::Lexicon,
    ) -> bool {
        let (name, category) = &self.marker_classes[class];
        // An intransitive marker cannot head the following frame Argument.
        lexicon.lexemes().values().any(|lexeme| {
            lexeme.category == *category
                && lexeme.properties.features.get("FrameMarker").map(|marker| marker.as_str()) == Some(name.as_str())
                && lexeme
                    .properties
                    .features
                    .get("PrepositionComplement")
                    .is_none_or(|complement| complement != "Intransitive")
        })
    }

    fn selected_expansion(
        &self,
        template: usize,
        head: FrameHead<'_>,
        values: &[FrameValue],
        lexicon: &::deckmaste_lexical::Lexicon,
        cache: &mut AdmissionCache,
    ) -> Result<(usize, Vec<Option<Summary>>), Error> {
        if !self.compiled {
            return Err(Error::EnvironmentRequired);
        }
        let summary = match (head, &self.productions[template].symbols[0]) {
            (FrameHead::Lexical(word), ::deckmaste_english_v3::Symbol::Lexical(pos)) => {
                cache.word(word, self, lexicon, *pos)?
            }
            (
                FrameHead::Constituent(reading),
                ::deckmaste_english_v3::Symbol::Nonterminal(category),
            ) if reading.category() == *category => cache.admit(reading, self, lexicon)?,
            _ => return Err(Error::Invalid("selected_frame", "selecting head category")),
        };
        let program = match summary.values.get(7).and_then(Option::as_ref) {
            Some(FeatureValue::Frame(index)) => self.frame_programs.get(*index).copied().flatten(),
            _ => None,
        }
        .ok_or(Error::Invalid(
            "selected_frame",
            "unsupported or absent declared frame",
        ))?;
        // Preserve the original expansion order and check every candidate for
        // this declaration/frame pair, without scanning unrelated productions.
        for &offset in self
            .expansion_index
            .get(&(template, program))
            .into_iter()
            .flatten()
        {
            let expansion = &self.expansions[offset];
            if expansion.layout.len() != values.len() {
                continue;
            }
            let mut summaries = vec![Some(summary.clone())];
            let mut valid = true;
            for &index in &expansion.order {
                if expansion.layout[index]
                    .summaries(&values[index], self, lexicon, &mut summaries, cache)
                    .is_err()
                {
                    valid = false;
                    break;
                }
            }
            if valid {
                return Ok((self.static_count + offset, summaries));
            }
        }
        Err(Error::Invalid(
            "selected_frame",
            "ordered declared frame values",
        ))
    }

    fn admit_selected(
        &self,
        template: usize,
        head: FrameHead<'_>,
        values: &[FrameValue],
        lexicon: &::deckmaste_lexical::Lexicon,
        cache: &mut AdmissionCache,
    ) -> Result<Summary, Error> {
        let (production, summaries) =
            self.selected_expansion(template, head, values, lexicon, cache)?;
        let rule = &self.rules[production];
        let mut state = State::new(rule.initial.clone());
        for (dot, summary) in summaries.iter().enumerate() {
            state = rule
                .advance(dot, &state, summary.as_ref())
                .ok_or(Error::Invalid("selected_frame", "feature admission"))?;
        }
        rule.complete(&state)
            .ok_or(Error::Invalid("selected_frame", "feature export"))
    }

    fn write_selected(
        &self,
        template: usize,
        head: FrameHead<'_>,
        values: &[FrameValue],
        lexicon: &::deckmaste_lexical::Lexicon,
        output: &mut Realization,
        cache: &mut AdmissionCache,
    ) -> Result<(), Error> {
        let (production, _) = self.selected_expansion(template, head, values, lexicon, cache)?;
        let expansion = &self.expansions[production - self.static_count];
        match head {
            FrameHead::Lexical(word) => output.word(word, lexicon)?,
            FrameHead::Constituent(reading) => reading.write(self, lexicon, output, cache)?,
        }
        for &index in &expansion.order {
            expansion.layout[index].write(&values[index], self, lexicon, output, cache)?;
        }
        Ok(())
    }

    fn materialize_selected(
        &self,
        production: usize,
        children: Vec<Value>,
    ) -> Result<Value, Error> {
        let expansion = self
            .expansions
            .get(
                production
                    .checked_sub(self.static_count)
                    .ok_or(Error::Internal)?,
            )
            .ok_or(Error::Internal)?;
        let mut children = children.into_iter();
        let head = children.next().ok_or(Error::Internal)?;
        let mut values = vec![None; expansion.layout.len()];
        for &index in &expansion.order {
            values[index] = Some(expansion.layout[index].materialize(&mut children)?);
        }
        let values = values
            .into_iter()
            .map(|value| value.ok_or(Error::Internal))
            .collect::<Result<Vec<_>, _>>()?;
        if children.next().is_some() {
            return Err(Error::Internal);
        }
        self.build_selected(expansion.template, head, values)
    }
}

impl FrameLayout {
    fn overt_slot(&self) -> Option<&Self> {
        match self {
            Self::Optional(None) | Self::Adjunct(_) => None,
            Self::Optional(Some(item)) => item.overt_slot(),
            _ => Some(self),
        }
    }
    fn postposed_object(
        &self,
        modifier: Option<Category>,
        extra: &[(usize, FeatureValue)],
    ) -> Option<Self> {
        match self {
            Self::Argument(category, requirements) => Some(Self::PostposedObject(
                *category,
                requirements.iter().chain(extra).cloned().collect(),
                modifier,
            )),
            Self::Optional(Some(item)) => Some(Self::Optional(Some(Box::new(
                item.postposed_object(modifier, extra)?,
            )))),
            _ => None,
        }
    }
    fn symbols(
        &self,
        symbols: &mut Vec<::deckmaste_english_v3::Symbol<Category>>,
        rule: &mut Rule,
        kind: &str,
    ) {
        fn append(
            symbols: &mut Vec<::deckmaste_english_v3::Symbol<Category>>,
            rule: &mut Rule,
            symbol: ::deckmaste_english_v3::Symbol<Category>,
            guard: FrameGuard,
        ) {
            symbols.push(::deckmaste_english_v3::Symbol::Literal(" ".into()));
            rule.guards.push(FrameGuard::Any);
            rule.literals.push(Some(" "));
            rule.checks.push(vec![]);
            rule.lexical_checks.push(vec![]);
            rule.release.push(vec![]);
            symbols.push(symbol);
            rule.guards.push(guard);
            rule.literals.push(None);
            rule.checks.push(vec![]);
            rule.lexical_checks.push(vec![]);
            rule.release.push(vec![]);
        }
        match self {
            Self::Adjunct(category) => {
                // The Adjunct Category owns its intervening separator.
                symbols.push(::deckmaste_english_v3::Symbol::Nonterminal(*category));
                rule.guards.push(FrameGuard::Any);
                rule.literals.push(None);
                rule.checks.push(vec![]);
                rule.lexical_checks.push(vec![]);
                rule.release.push(vec![]);
            }
            Self::PostposedObject(category, requirements, modifier) => {
                if let Some(modifier) = modifier {
                    Self::Argument(*modifier, vec![]).symbols(symbols, rule, kind);
                }
                Self::Argument(*category, requirements.clone()).symbols(symbols, rule, kind);
            }
            Self::Argument(category, requirements) => append(
                symbols,
                rule,
                ::deckmaste_english_v3::Symbol::Nonterminal(*category),
                FrameGuard::Features(requirements.clone()),
            ),
            Self::Marker(class, pos) => append(
                symbols,
                rule,
                ::deckmaste_english_v3::Symbol::Lexical(*pos),
                FrameGuard::Marker(*class, frame_marker_licence(kind, *pos)),
            ),
            Self::Marked(class, pos, category, requirements) => {
                Self::Marker(*class, *pos).symbols(symbols, rule, kind);
                Self::Argument(*category, requirements.clone()).symbols(symbols, rule, kind);
            }
            Self::Optional(Some(item)) => item.symbols(symbols, rule, kind),
            Self::Optional(None) => {}
        }
    }
    fn summaries(
        &self,
        value: &FrameValue,
        grammar: &Grammar,
        lexicon: &::deckmaste_lexical::Lexicon,
        output: &mut Vec<Option<Summary>>,
        cache: &mut AdmissionCache,
    ) -> Result<(), Error> {
        match (self, value) {
            (
                Self::PostposedObject(category, _, modifier_category),
                FrameValue::PostposedObject { argument, modifier },
            ) => {
                match (modifier_category, modifier) {
                    (Some(category), Some(value)) if value.category() == *category => {
                        output.push(None);
                        output.push(Some(cache.admit(value, grammar, lexicon)?));
                    }
                    (None, None) => {}
                    _ => return Err(Error::Invalid("selected_frame", "postposed modifier")),
                }
                if argument.category() != *category {
                    return Err(Error::Invalid(
                        "selected_frame",
                        "postposed Object category",
                    ));
                }
                output.push(None);
                output.push(Some(cache.admit(argument, grammar, lexicon)?));
                Ok(())
            }
            (Self::Adjunct(category), FrameValue::Adjunct(reading))
                if reading.category() == *category =>
            {
                output.push(Some(cache.admit(reading, grammar, lexicon)?));
                Ok(())
            }
            (Self::Argument(category, _), FrameValue::Argument(reading))
                if reading.category() == *category =>
            {
                output.push(None);
                output.push(Some(cache.admit(reading, grammar, lexicon)?));
                Ok(())
            }
            (Self::Marker(class, pos), FrameValue::Marker(word)) => {
                let summary = cache.word(word, grammar, lexicon, *pos)?;
                if summary.marker_class != Some(*class) {
                    return Err(Error::Invalid("selected_frame", "declared marker class"));
                }
                output.push(None);
                output.push(Some(summary));
                Ok(())
            }
            (Self::Marked(class, pos, category, _), FrameValue::Marked { marker, argument }) => {
                let summary = cache.word(marker, grammar, lexicon, *pos)?;
                if summary.marker_class != Some(*class) {
                    return Err(Error::Invalid("selected_frame", "declared marker class"));
                }
                output.push(None);
                output.push(Some(summary));
                if argument.category() != *category {
                    return Err(Error::Invalid(
                        "selected_frame",
                        "slot category or optional realization",
                    ));
                }
                output.push(None);
                output.push(Some(cache.admit(argument, grammar, lexicon)?));
                Ok(())
            }
            (Self::Optional(None), FrameValue::Optional(None)) => Ok(()),
            (Self::Optional(Some(item)), FrameValue::Optional(Some(value))) => {
                item.summaries(value, grammar, lexicon, output, cache)
            }
            _ => Err(Error::Invalid(
                "selected_frame",
                "slot category or optional realization",
            )),
        }
    }
    fn write(
        &self,
        value: &FrameValue,
        grammar: &Grammar,
        lexicon: &::deckmaste_lexical::Lexicon,
        output: &mut Realization,
        cache: &mut AdmissionCache,
    ) -> Result<(), Error> {
        match (self, value) {
            (
                Self::PostposedObject(_, _, _),
                FrameValue::PostposedObject { argument, modifier },
            ) => {
                if let Some(modifier) = modifier {
                    output.push_str(" ");
                    modifier.write(grammar, lexicon, output, cache)?;
                }
                output.push_str(" ");
                argument.write(grammar, lexicon, output, cache)
            }
            (Self::Adjunct(_), FrameValue::Adjunct(reading)) => {
                reading.write(grammar, lexicon, output, cache)
            }
            (Self::Argument(_, _), FrameValue::Argument(reading)) => {
                output.push_str(" ");
                reading.write(grammar, lexicon, output, cache)
            }
            (Self::Marker(_, _), FrameValue::Marker(word)) => {
                output.push_str(" ");
                output.word(word, lexicon)
            }
            (Self::Marked(_, _, _, _), FrameValue::Marked { marker, argument }) => {
                output.push_str(" ");
                output.word(marker, lexicon)?;
                output.push_str(" ");
                argument.write(grammar, lexicon, output, cache)
            }
            (Self::Optional(None), FrameValue::Optional(None)) => Ok(()),
            (Self::Optional(Some(item)), FrameValue::Optional(Some(value))) => {
                item.write(value, grammar, lexicon, output, cache)
            }
            _ => Err(Error::Internal),
        }
    }
    fn materialize(&self, children: &mut impl Iterator<Item = Value>) -> Result<FrameValue, Error> {
        fn next(children: &mut impl Iterator<Item = Value>) -> Result<Value, Error> {
            if !matches!(children.next(), Some(Value::Literal)) {
                return Err(Error::Internal);
            }
            children.next().ok_or(Error::Internal)
        }
        match self {
            Self::PostposedObject(_, _, modifier) => Ok(FrameValue::PostposedObject {
                modifier: modifier
                    .map(|_| next(children).and_then(Value::node).map(Box::new))
                    .transpose()?,
                argument: Box::new(next(children)?.node()?),
            }),
            Self::Adjunct(_) => Ok(FrameValue::Adjunct(Box::new(
                children.next().ok_or(Error::Internal)?.node()?,
            ))),
            Self::Argument(_, _) => Ok(FrameValue::Argument(Box::new(next(children)?.node()?))),
            Self::Marker(_, _) => Ok(FrameValue::Marker(next(children)?.word()?)),
            Self::Marked(_, _, _, _) => Ok(FrameValue::Marked {
                marker: next(children)?.word()?,
                argument: Box::new(next(children)?.node()?),
            }),
            Self::Optional(None) => Ok(FrameValue::Optional(None)),
            Self::Optional(Some(item)) => Ok(FrameValue::Optional(Some(Box::new(
                item.materialize(children)?,
            )))),
        }
    }
}

impl FrameValue {
    fn is_postposed(&self) -> bool {
        match self {
            Self::PostposedObject { .. } => true,
            Self::Optional(Some(item)) => item.is_postposed(),
            _ => false,
        }
    }
    fn visit(&self, visitor: &mut impl FnMut(&Reading)) -> Result<(), Error> {
        match self {
            Self::PostposedObject { argument, modifier } => {
                if let Some(modifier) = modifier {
                    modifier.visit(visitor)?;
                }
                argument.visit(visitor)
            }
            Self::Argument(reading)
            | Self::Adjunct(reading)
            | Self::Marked {
                argument: reading, ..
            } => reading.visit(visitor),
            Self::Optional(Some(item)) => item.visit(visitor),
            _ => Ok(()),
        }
    }
    fn visit_words(&self, visitor: &mut impl FnMut(&Word)) -> Result<(), Error> {
        match self {
            Self::PostposedObject { argument, modifier } => {
                if let Some(modifier) = modifier {
                    modifier.visit_words(visitor)?;
                }
                argument.visit_words(visitor)
            }
            Self::Argument(reading) | Self::Adjunct(reading) => reading.visit_words(visitor),
            Self::Marker(word) => {
                visitor(word);
                Ok(())
            }
            Self::Marked { marker, argument } => {
                visitor(marker);
                argument.visit_words(visitor)
            }
            Self::Optional(Some(item)) => item.visit_words(visitor),
            Self::Optional(None) => Ok(()),
        }
    }
}
